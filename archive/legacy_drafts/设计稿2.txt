    document.querySelectorAll('.sector-option').forEach(btn => {

        btn.style.borderColor  = match ? 'var(--color-primary)' : 'var(--border-color)';
        btn.style.color        = match ? 'var(--color-primary)' : 'var(--text-main)';

    });

function setupBottomNav() {
    document.querySelectorAll('.nav-item').forEach(item => {
        item.addEventListener('click', () => {
            document.querySelectorAll('.nav-item').forEach(nav => {
                nav.style.color = 'var(--text-muted)';
                const icon = nav.querySelector('i');
                if (icon) icon.className = icon.className.replace('ph-fill', 'ph');
                nav.classList.remove('active-nav');
            });
            item.style.color = 'var(--color-primary)';
            item.classList.add('active-nav');
            const icon = item.querySelector('i');
            if (icon) icon.className = icon.className.replace(/\bph\b/, 'ph-fill');

            if (item.dataset.target === 'view-library') {
                document.getElementById('library-dot').style.opacity = '0';
            }
            document.querySelectorAll('.view-content').forEach(v => v.classList.remove('active'));
            document.getElementById(item.dataset.target).classList.add('active');
        });
    });
}

async function generateAICards(sector) {
    if (isGenerating) return;
    isGenerating = true;
    currentCards = [];
    
    const container = document.getElementById('card-container');
    container.innerHTML = '';
    
    // Show AI status
    document.getElementById('ai-status-bar').classList.remove('hidden');
    document.getElementById('refresh-icon').style.color = 'var(--color-primary)';
    document.getElementById('splash-text') && (document.getElementById('splash-text').innerText = 'Generating Insights...');

    // Show skeleton loading card
    container.innerHTML = `
        <div class="skeleton-card">
            <div class="flex justify-between mb-4">
                <div class="skeleton-line w-24"></div>
                <div class="skeleton-line w-16"></div>
            </div>
            <div class="skeleton-line w-full mb-1"></div>
            <div class="skeleton-line w-4/5 mb-6"></div>
            <div class="skeleton-box mb-3"></div>
            <div class="skeleton-box mb-3"></div>
            <div class="skeleton-box"></div>
        </div>
        <div class="absolute bottom-6 left-0 right-0 flex justify-center">
            <div class="ai-status">
                <div class="ai-dot"></div>
                <span class="font-mono-special text-[11px]">AI 正在分析 ${sector} 赛道...</span>
            </div>
        </div>
    `;

    const sectorMap = {
        'SaaS商业化': 'SaaS商业化/出海',
        '教育新媒体': '职业教育/新媒体获客',
        '本地生活电商': '本地生活/电商变现',
    };
    const fullSector = sectorMap[sector] || sector;

    const prompt = `你是一个顶级AI商业情报分析师。请为"${fullSector}"赛道生成3条高质量的AI商业情报卡片，格式为JSON数组。

每条情报必须包含：
- id: 唯一数字ID（从${Date.now()}开始递增）
- type: "warning"（颠覆预警）| "opportunity"（增长机遇）| "normal"（模式观察）
- tag: 对应中文标签（"颠覆预警"/"增长机遇"/"模式观察"）
- heat: 匹配度百分比，如"匹配度 96%"
- title: 具体的情报标题（20-40字，要有具体数据或事件，有冲击力）
- source: 来源描述（如"a16z深度播客精华提炼"、"开发者大会核心QA音频拆解"）
- sourceType: "video" | "audio" | "article"
- sourceTime: 如"45:12" 或 "阅读时长 12min"
- useCase: 具体落地场景描述（2-3句话）
- monetization: 商业模式与盈利方式描述（2-3句话）
- roi: ROI数据与效能提升描述（包含具体数字，2-3句话）
- friction: 落地阻碍与风险（2-3句话）
- action: 具体的战略级行动建议（2-3句话，要可执行）

要求：
1. 内容必须真实反映${fullSector}赛道的最新商业动态
2. 数据要具体（百分比、金额、时间等）
3. 行动建议要可落地执行
4. 不同卡片的type要多样化
5. 只输出JSON数组，不要任何markdown或解释文字

示例格式：
[{"id":1,"type":"warning","tag":"颠覆预警","heat":"匹配度 98%","title":"...","source":"...","sourceType":"video","sourceTime":"45:12","useCase":"...","monetization":"...","roi":"...","friction":"...","action":"..."}]`;

    try {
        const response = await fetch("https://api.anthropic.com/v1/messages", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                model: "claude-sonnet-4-20250514",
                max_tokens: 1000,
                messages: [{ role: "user", content: prompt }]
            })
        });

        const data = await response.json();
        const raw = data.content?.[0]?.text || '[]';
        
        // Clean up potential markdown fences
        const clean = raw.replace(/```json\n?/g, '').replace(/```\n?/g, '').trim();
        const cards = JSON.parse(clean);

        if (!Array.isArray(cards) || cards.length === 0) throw new Error('No cards returned');

        currentCards = cards;
        container.innerHTML = '';
        renderCards(cards);
    } catch (err) {
        console.error('AI generation failed:', err);
        // Fallback to built-in cards
        currentCards = getFallbackCards(sector);
        container.innerHTML = '';
        renderCards(currentCards);
    } finally {
        isGenerating = false;
        document.getElementById('ai-status-bar').classList.add('hidden');
        document.getElementById('refresh-icon').style.color = 'var(--text-muted)';
    }
}

function refreshCards() {
    const sector = localStorage.getItem('insightx_sector') || 'SaaS商业化';
    generateAICards(sector);
}

// ═══════════════════════════════════════════════
//  DRAG LOGIC
// ═══════════════════════════════════════════════
function makeCardDraggable(card, data) {
    let isDragging = false, startX = 0, startY = 0, currentX = 0, currentY = 0;
    const badgeLike = card.querySelector('.badge.like');
    const badgeNope = card.querySelector('.badge.nope');

    const startDrag = (e) => {
        if (e.target.closest('button')) return;
        isDragging = true;
        card.classList.add('dragging');
        startX = e.type.includes('mouse') ? e.pageX : e.touches[0].pageX;
        startY = e.type.includes('mouse') ? e.pageY : e.touches[0].pageY;
    };
    const onDrag = (e) => {
        if (!isDragging) return;
        const x = e.type.includes('mouse') ? e.pageX : e.touches[0].pageX;
        const y = e.type.includes('mouse') ? e.pageY : e.touches[0].pageY;
        currentX = x - startX; currentY = y - startY;
        card.style.transform = `translate(${currentX}px, ${currentY}px) rotate(${currentX * 0.04}deg)`;
        if (currentX > 0) {
            badgeLike.style.opacity = Math.min(currentX / 100, 1);
            badgeNope.style.opacity = 0;
        } else {
            badgeNope.style.opacity = Math.min(Math.abs(currentX) / 100, 1);
            badgeLike.style.opacity = 0;
        }
    };
    const stopDrag = () => {
        if (!isDragging) return;
        isDragging = false;
        card.classList.remove('dragging');
        if (currentX > 100) {
            card.style.transform = `translate(${window.innerWidth}px, ${currentY}px) rotate(30deg)`;
            card.style.opacity = 0;
            saveToLibrary(data);
            setTimeout(() => nextCard(card), 300);
        } else if (currentX < -100) {
            card.style.transform = `translate(-${window.innerWidth}px, ${currentY}px) rotate(-30deg)`;
            card.style.opacity = 0;
            setTimeout(() => nextCard(card), 300);
        } else {
            card.style.transform = `translate(0px, 0px) rotate(0deg)`;
            badgeLike.style.opacity = 0;
            badgeNope.style.opacity = 0;
        }
    };

    card.addEventListener('mousedown', startDrag);
    card.addEventListener('touchstart', startDrag, { passive: true });
    document.addEventListener('mousemove', onDrag);
    document.addEventListener('touchmove', onDrag, { passive: true });
    document.addEventListener('mouseup', stopDrag);
    document.addEventListener('touchend', stopDrag);
}

function nextCard(oldCard) {
    oldCard.remove();
    const container = document.getElementById('card-container');
    const cards = Array.from(container.querySelectorAll('.insight-card'));
    if (cards.length > 0) {
        cards.forEach((card, idx) => {
            const rev = cards.length - 1 - idx;
            card.style.zIndex = idx;
            card.style.transform = `scale(${1 - rev * 0.04}) translateY(${rev * 12}px)`;
            card.style.opacity = 1 - rev * 0.15;
        });
        const topCardIndex = currentCards.length - cards.length;
        makeCardDraggable(cards[cards.length - 1], currentCards[topCardIndex]);
    } else {
        container.innerHTML = `
            <div class="h-full flex flex-col justify-center items-center text-center px-8">
                <i class="ph-fill ph-check-circle text-6xl mb-5 shadow-[0_0_30px_rgba(129,161,193,0.3)] rounded-full" style="color: var(--color-primary);"></i>
                <p class="font-bold text-lg tracking-widest">INBOX ZERO</p>
                <p class="text-sm mt-3 opacity-60">今日高价值情报已批阅完毕</p>
                <button class="mt-8 border px-8 py-3 rounded-xl text-sm font-bold active:scale-95 transition-transform"
                        style="border-color: var(--color-primary); color: var(--color-primary); background: rgba(129,161,193,0.05);"
                        onclick="refreshCards()">
                    <i class="ph ph-sparkle mr-1"></i> AI 刷新情报流
                </button>
            </div>
        `;
    }
}

// ═══════════════════════════════════════════════
//  LIBRARY
// ═══════════════════════════════════════════════
function saveToLibrary(data) {
    if (savedItems.find(i => i.id === data.id)) return;
    savedItems.unshift({ ...data, pinned: false });
    updateLibraryUI();
    document.getElementById('library-dot').style.opacity = '1';
}

function togglePin(id, e) {
    e.stopPropagation();
    const idx = savedItems.findIndex(i => i.id === id);
    if (idx > -1) { savedItems[idx].pinned = !savedItems[idx].pinned; updateLibraryUI(); }
}

function deleteItem(id, e) {
    e.stopPropagation();
    savedItems = savedItems.filter(i => i.id !== id);
    updateLibraryUI();
}

function updateLibraryUI() {
    document.getElementById('saved-count').innerText = `${savedItems.length} 条归档`;
    const list = document.getElementById('library-list');
    const empty = document.getElementById('empty-library');
    Array.from(list.children).forEach(c => { if (c.id !== 'empty-library') c.remove(); });

    if (savedItems.length === 0) { empty.style.display = 'flex'; return; }
    empty.style.display = 'none';

    const sorted = [...savedItems].sort((a, b) => (b.pinned === true) - (a.pinned === true));
    sorted.forEach(data => {
        const tagColor = data.type === 'warning'     ? 'var(--color-warning)'   :
                         data.type === 'opportunity' ? 'var(--color-success)'   :
                                                       'var(--color-highlight)';
        const el = document.createElement('div');
        el.className = 'p-4 rounded-xl border flex flex-col gap-2 cursor-pointer transition-colors relative active:scale-[0.98]';
        el.style.backgroundColor = data.pinned ? 'rgba(208,135,112,0.05)' : 'var(--bg-card)';
        el.style.borderColor     = data.pinned ? 'var(--color-warning)'   : 'var(--border-color)';
        el.setAttribute('onclick', `openLibraryDetail(${data.id})`);
        el.innerHTML = `
            <div class="flex justify-between items-start">
                <div class="flex items-center gap-2">
                    ${data.pinned ? `<i class="ph-fill ph-push-pin text-[14px]" style="color: var(--color-warning);"></i>` : ''}
                    <span class="text-[10px] border px-2 py-0.5 rounded font-bold tracking-wider" style="color: ${tagColor}; border-color: ${tagColor};">${data.tag}</span>
                </div>
                <div class="flex gap-4 text-lg opacity-80 pl-4">
                    <i class="${data.pinned ? 'ph-fill' : 'ph'} ph-push-pin transition-colors active:scale-90"
                       style="color: ${data.pinned ? 'var(--color-warning)' : 'var(--text-muted)'};"
                       onclick="togglePin(${data.id}, event)"></i>
                    <i class="ph ph-trash transition-colors active:scale-90 hover:text-red-400"
                       style="color: var(--text-muted);"
                       onclick="deleteItem(${data.id}, event)"></i>
                </div>
            </div>
            <h4 class="text-[14px] font-bold mt-1.5 leading-snug pr-2">${data.title}</h4>
            <p class="text-[11px] opacity-60 mt-1 flex items-center gap-1.5 font-bold"><i class="ph ph-link text-sm"></i> ${data.source}</p>
        `;
        list.appendChild(el);
    });
}
