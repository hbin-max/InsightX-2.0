function setupBottomNav() {
    document.querySelectorAll('.nav-item').forEach(item => {
        item.addEventListener('click', () => {
            document.querySelectorAll('.nav-item').forEach(nav => {
                nav.style.color = 'var(--text-muted)';
                const icon = nav.querySelector('i');
                if (icon) icon.className = icon.className.replace('ph-fill', 'ph');
                nav.classList.remove('active-nav');
            });
            item.style.color = 'var(--color-primary)';
            item.classList.add('active-nav');
            const icon = item.querySelector('i');

        });
    });
}

function makeCardDraggable(card, data) {
    let isDragging = false, startX = 0, startY = 0, currentX = 0, currentY = 0;
    const badgeLike = card.querySelector('.badge.like');
    const badgeNope = card.querySelector('.badge.nope');

    const startDrag = (e) => {
        if (e.target.closest('button')) return;
        isDragging = true;
        card.classList.add('dragging');
        startX = e.type.includes('mouse') ? e.pageX : e.touches[0].pageX;
        startY = e.type.includes('mouse') ? e.pageY : e.touches[0].pageY;
    };
    const onDrag = (e) => {
        if (!isDragging) return;
        const x = e.type.includes('mouse') ? e.pageX : e.touches[0].pageX;
        const y = e.type.includes('mouse') ? e.pageY : e.touches[0].pageY;
        currentX = x - startX; currentY = y - startY;
        card.style.transform = `translate(${currentX}px, ${currentY}px) rotate(${currentX * 0.04}deg)`;
        if (currentX > 0) {
            badgeLike.style.opacity = Math.min(currentX / 100, 1);
            badgeNope.style.opacity = 0;
        } else {
            badgeNope.style.opacity = Math.min(Math.abs(currentX) / 100, 1);
            badgeLike.style.opacity = 0;
        }
    };
    const stopDrag = () => {
        if (!isDragging) return;
        isDragging = false;
        card.classList.remove('dragging');
        if (currentX > 100) {
            card.style.transform = `translate(${window.innerWidth}px, ${currentY}px) rotate(30deg)`;
            card.style.opacity = 0;
            saveToLibrary(data);
            setTimeout(() => nextCard(card), 300);
        } else if (currentX < -100) {
            card.style.transform = `translate(-${window.innerWidth}px, ${currentY}px) rotate(-30deg)`;
            card.style.opacity = 0;
            setTimeout(() => nextCard(card), 300);
        } else {
            card.style.transform = `translate(0px, 0px) rotate(0deg)`;
            badgeLike.style.opacity = 0; badgeNope.style.opacity = 0;
        }
    };

    card.addEventListener('mousedown', startDrag);
    card.addEventListener('touchstart', startDrag, { passive: true });
    document.addEventListener('mousemove', onDrag);
    document.addEventListener('touchmove', onDrag, { passive: true });
    document.addEventListener('mouseup', stopDrag);
    document.addEventListener('touchend', stopDrag);
}

function nextCard(oldCard) {
    oldCard.remove();
    const container = document.getElementById('card-container');
    const cards = Array.from(container.querySelectorAll('.insight-card'));
    if (cards.length > 0) {
        cards.forEach((card, idx) => {
            const rev = cards.length - 1 - idx;
            card.style.zIndex = idx;
            card.style.transform = `scale(${1 - rev * 0.04}) translateY(${rev * 12}px)`;
            card.style.opacity = 1 - rev * 0.15;
        });

// ═══════════════════════════════════════════════
//  LIBRARY & ADVANCED FLIP ANIMATIONS
// ═══════════════════════════════════════════════
function updateLibraryUI() {
    document.getElementById('saved-count').innerText = `${savedItems.length} 条归档`;
    const list = document.getElementById('library-list');
    const empty = document.getElementById('empty-library');

    // 移除所有未被标记为删除的节点
    Array.from(list.children).forEach(c => {
        if (!c.classList.contains('deleting')) c.remove();
    });

    if (savedItems.length === 0) {
        empty.style.display = 'flex';
        setTimeout(() => empty.style.opacity = '0.5', 50);
        return;
    }
    empty.style.opacity = '0';
    setTimeout(() => empty.style.display = 'none', 300);

    const sorted = [...savedItems].sort((a, b) => (b.pinned === true) - (a.pinned === true));
    
    sorted.forEach(data => {
        // 如果该节点正在被删除动画中，跳过重绘
        if(document.querySelector(`.lib-item[data-id="${data.id}"].deleting`)) return;

        const tagColor = data.type === 'warning' ? 'var(--color-warning)' : data.type === 'opportunity' ? 'var(--color-success)' : 'var(--color-highlight)';
        const el = document.createElement('div');
        el.className = 'lib-item p-4 rounded-xl border flex flex-col gap-2 cursor-pointer relative';
        el.dataset.id = data.id; // 用于FLIP关联
        el.style.backgroundColor = data.pinned ? 'rgba(208,135,112,0.05)' : 'var(--bg-card)';
        el.style.borderColor     = data.pinned ? 'var(--color-warning)'   : 'var(--border-color)';
        el.setAttribute('onclick', `openLibraryDetail(${data.id})`);

        el.innerHTML = `
            <div class="flex justify-between items-start">
                <div class="flex items-center gap-2">
                    ${data.pinned ? `<i class="ph-fill ph-push-pin text-[14px]" style="color: var(--color-warning);"></i>` : ''}
                    <span class="text-[10px] border px-2 py-0.5 rounded font-bold tracking-wider" style="color: ${tagColor}; border-color: ${tagColor};">${data.tag}</span>
                </div>
                <div class="flex gap-4 text-lg opacity-80 pl-4 relative z-10">
                    <i class="${data.pinned ? 'ph-fill' : 'ph'} ph-push-pin transition-all hover:scale-125"
                       style="color: ${data.pinned ? 'var(--color-warning)' : 'var(--text-muted)'};"
                       onclick="togglePin(${data.id}, event)"></i>
                    <i class="ph ph-trash transition-all hover:scale-125 hover:text-red-400"
                       style="color: var(--text-muted);"
                       onclick="deleteItem(${data.id}, event)"></i>
                </div>
            </div>
            <h4 class="text-[14px] font-bold mt-1.5 leading-snug pr-2">${data.title}</h4>
            <p class="text-[11px] opacity-60 mt-1 flex items-center gap-1.5 font-bold"><i class="ph ph-link text-sm"></i> ${data.source}</p>
        `;
        list.appendChild(el);
    });
}

function saveToLibrary(data) {
    if (savedItems.find(i => i.id === data.id)) return;
    savedItems.unshift({ ...data, pinned: false });
    
    updateLibraryUI();
    document.getElementById('library-dot').style.opacity = '1';

    // Slide in 动画
    const newItem = document.querySelector(`.lib-item[data-id="${data.id}"]`);
    if(newItem) {
        newItem.style.transition = 'none';
        newItem.style.opacity = '0';
        newItem.style.transform = 'translateY(-20px) scale(0.95)';
        requestAnimationFrame(() => {
            requestAnimationFrame(() => {
                newItem.style.transition = 'opacity 0.4s ease, transform 0.4s cubic-bezier(0.2, 0.8, 0.2, 1)';
                newItem.style.opacity = '1';
                newItem.style.transform = 'translateY(0) scale(1)';
            });
        });
    }
}

// FLIP 置顶/取消置顶平滑动画
function togglePin(id, e) {
    if(e) e.stopPropagation();
    const idx = savedItems.findIndex(i => i.id === id);
    if (idx > -1) {
        // [F]irst: 记录当前所有位置
        const list = document.getElementById('library-list');
        const items = Array.from(list.children).filter(c => !c.classList.contains('deleting'));
        const firstRects = new Map();
        items.forEach(item => firstRects.set(item.dataset.id, item.getBoundingClientRect()));

        // Update state and Render
        savedItems[idx].pinned = !savedItems[idx].pinned;
        updateLibraryUI();

        // [L]ast & [I]nvert
        const newItems = Array.from(list.children).filter(c => !c.classList.contains('deleting'));
        newItems.forEach(item => {
            const firstRect = firstRects.get(item.dataset.id);
            if (firstRect) {
                const lastRect = item.getBoundingClientRect();
                const deltaY = firstRect.top - lastRect.top;
                if (deltaY !== 0) {
                    item.style.transform = `translateY(${deltaY}px)`;
                    item.style.transition = 'none'; // 瞬间归位

                    // [P]lay
                    requestAnimationFrame(() => {
                        item.getBoundingClientRect(); // 强制重绘
                        item.style.transition = 'transform 0.45s cubic-bezier(0.34, 1.56, 0.64, 1), background-color 0.4s ease, border-color 0.4s ease';
                        item.style.transform = 'translateY(0)';
                    });
                }
            }
        });
    }
}

// 侧滑+平滑折叠删除动画
function deleteItem(id, e) {
    if(e) e.stopPropagation();
    const itemEl = document.querySelector(`.lib-item[data-id="${id}"]`);
    
    if (itemEl) {
        itemEl.classList.add('deleting'); // 标记为正在删除
        
        // 1. 向右滑出并透明
        itemEl.style.transition = 'transform 0.35s cubic-bezier(0.4, 0, 0.2, 1), opacity 0.3s ease';
        itemEl.style.transform = 'translateX(120%)';
        itemEl.style.opacity = '0';

        setTimeout(() => {
            // 2. 高度平滑折叠，带动下方元素上移
            const currentHeight = itemEl.offsetHeight;
            itemEl.style.maxHeight = currentHeight + 'px'; // 设置显式高度以便计算
            itemEl.offsetHeight; // 强制重绘
            
            itemEl.style.transition = 'max-height 0.35s cubic-bezier(0.4, 0, 0.2, 1), margin 0.35s ease, padding 0.35s ease, border-width 0.35s ease';
            itemEl.style.maxHeight = '0px';
            itemEl.style.marginTop = '0px';
            itemEl.style.marginBottom = '0px';
            itemEl.style.paddingTop = '0px';
            itemEl.style.paddingBottom = '0px';
            itemEl.style.borderWidth = '0px';

            setTimeout(() => {
                // 3. 彻底移除DOM并更新状态
                savedItems = savedItems.filter(i => i.id !== id);
                itemEl.remove();
                updateLibraryUI();
            }, 350);
        }, 300);
    } else {
        savedItems = savedItems.filter(i => i.id !== id);
        updateLibraryUI();
    }
}

function makeLibraryCardDraggable(card) {
    let isDragging = false, startX = 0, startY = 0, currentX = 0, currentY = 0;
    const badgeLike = card.querySelector('.badge.like');
    const badgeNope = card.querySelector('.badge.nope');

    const startDrag = (e) => {
        if (e.target.closest('button')) return;
        isDragging = true;
        card.classList.add('dragging');
        startX = e.type.includes('mouse') ? e.pageX : e.touches[0].pageX;
        startY = e.type.includes('mouse') ? e.pageY : e.touches[0].pageY;
    };
    const onDrag = (e) => {
        if (!isDragging) return;
        const x = e.type.includes('mouse') ? e.pageX : e.touches[0].pageX;
        const y = e.type.includes('mouse') ? e.pageY : e.touches[0].pageY;
        currentX = x - startX; currentY = y - startY;
        card.style.transform = `translate(${currentX}px, ${currentY}px) rotate(${currentX * 0.04}deg)`;
        if (currentX > 0) {
            badgeLike.style.opacity = Math.min(currentX / 100, 1);
            badgeNope.style.opacity = 0;
        } else {
            badgeNope.style.opacity = Math.min(Math.abs(currentX) / 100, 1);
            badgeLike.style.opacity = 0;
        }
    };
    const stopDrag = () => {
        if (!isDragging) return;
        isDragging = false;
        card.classList.remove('dragging');
        if (Math.abs(currentX) > 100) {
            const sign = currentX > 0 ? 1 : -1;
            card.style.transform = `translate(${sign * window.innerWidth}px, ${currentY}px) rotate(${sign * 30}deg)`;
            card.style.opacity = 0;
            setTimeout(closeLibraryDetail, 300);
        } else {
            card.style.transform = `translate(0px, 0px) rotate(0deg)`;
            badgeLike.style.opacity = 0;
            badgeNope.style.opacity = 0;
        }
    };

    card.addEventListener('mousedown', startDrag);
    card.addEventListener('touchstart', startDrag, { passive: true });
    document.addEventListener('mousemove', onDrag);
    document.addEventListener('touchmove', onDrag, { passive: true });
    document.addEventListener('mouseup', stopDrag);
    document.addEventListener('touchend', stopDrag);
}
