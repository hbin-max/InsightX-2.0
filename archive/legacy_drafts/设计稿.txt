    function makeCardDraggable(card, data) {
        let isDragging = false, startX = 0, startY = 0, currentX = 0, currentY = 0;
        const badgeLike = card.querySelector('.badge.like');
        const badgeNope = card.querySelector('.badge.nope');

        const startDrag = (e) => {
            if (e.target.closest('button')) return;
            isDragging = true;
            card.classList.add('dragging');
            startX = e.type.includes('mouse') ? e.pageX : e.touches[0].pageX;
            startY = e.type.includes('mouse') ? e.pageY : e.touches[0].pageY;
        };

        const onDrag = (e) => {
            if (!isDragging) return;
            const x = e.type.includes('mouse') ? e.pageX : e.touches[0].pageX;
            const y = e.type.includes('mouse') ? e.pageY : e.touches[0].pageY;
            currentX = x - startX; currentY = y - startY;
            
            card.style.transform = `translate(${currentX}px, ${currentY}px) rotate(${currentX * 0.04}deg)`;

            if (currentX > 0) {
                badgeLike.style.opacity = Math.min(currentX / 100, 1);
                badgeNope.style.opacity = 0;
            } else {
                badgeNope.style.opacity = Math.min(Math.abs(currentX) / 100, 1);
                badgeLike.style.opacity = 0;
            }
        };

        const stopDrag = () => {
            if (!isDragging) return;
            isDragging = false;
            card.classList.remove('dragging');

            if (currentX > 100) {
                card.style.transform = `translate(${window.innerWidth}px, ${currentY}px) rotate(30deg)`;
                card.style.opacity = 0;
                saveToLibrary(data);
                setTimeout(() => nextCard(card), 300);
            } else if (currentX < -100) {
                card.style.transform = `translate(-${window.innerWidth}px, ${currentY}px) rotate(-30deg)`;
                card.style.opacity = 0;
                setTimeout(() => nextCard(card), 300);
            } else {
                card.style.transform = `translate(0px, 0px) rotate(0deg)`;
                badgeLike.style.opacity = 0; badgeNope.style.opacity = 0;
            }
        };

        card.addEventListener('mousedown', startDrag);
        card.addEventListener('touchstart', startDrag, {passive: true});
        document.addEventListener('mousemove', onDrag);
        document.addEventListener('touchmove', onDrag, {passive: true});
        document.addEventListener('mouseup', stopDrag);
        document.addEventListener('touchend', stopDrag);
    }

    // ====== Req 2: 深度智库 (Library) 交互逻辑 (支持置顶与删除) ======
    function saveToLibrary(data) {
        if(savedItems.find(item => item.id === data.id)) return;
        
        // 新增数据附带 pinned 属性
        savedItems.unshift({...data, pinned: false});
        updateLibraryUI();
        document.getElementById('library-dot').style.opacity = '1';
    }

    function togglePin(id, e) {
        e.stopPropagation();
        const index = savedItems.findIndex(item => item.id === id);
        if(index > -1) {
            savedItems[index].pinned = !savedItems[index].pinned;
            updateLibraryUI();
        }
    }

    function deleteItem(id, e) {
        e.stopPropagation();
        savedItems = savedItems.filter(item => item.id !== id);
        updateLibraryUI();
    }

    function updateLibraryUI() {
        document.getElementById('saved-count').innerText = `${savedItems.length} 条归档`;
        const list = document.getElementById('library-list');
        const emptyState = document.getElementById('empty-library');
        
        // 【修复点】：只移除动态生成的卡片，绝对不能删掉空状态节点
        Array.from(list.children).forEach(child => {
            if (child.id !== 'empty-library') {
                child.remove();
            }
        });
        
        if (savedItems.length === 0) {
            emptyState.style.display = 'flex';
            return;
        }
        
        emptyState.style.display = 'none';

        // 排序规则：置顶的在最上方
        const sortedItems = [...savedItems].sort((a, b) => (b.pinned === true) - (a.pinned === true));

        sortedItems.forEach(data => {
            const itemEl = document.createElement('div');
            // 如果置顶，边框颜色高亮
            const borderColor = data.pinned ? 'var(--color-warning)' : 'var(--border-color)';
            const bgColor = data.pinned ? 'rgba(208,135,112,0.05)' : 'var(--bg-card)';
            
            // 【新增】：添加 onclick 事件以呼出浮窗卡片
            itemEl.className = 'p-4 rounded-xl border flex flex-col gap-2 cursor-pointer transition-colors relative';
            itemEl.style.backgroundColor = bgColor;
            itemEl.style.borderColor = borderColor;
            itemEl.setAttribute('onclick', `openLibraryDetail(${data.id})`);
            
            itemEl.innerHTML = `
                <div class="flex justify-between items-start">
                    <div class="flex items-center gap-2">
                        ${data.pinned ? `<i class="ph-fill ph-push-pin text-[14px]" style="color: var(--color-warning);"></i>` : ''}
                        <span class="text-[10px] border px-2 py-0.5 rounded font-bold tracking-wider" style="color: var(--color-success); border-color: var(--color-success);">${data.tag}</span>
                    </div>
                    
                    <!-- Req 2: 动作按钮栏 (置顶 & 删除) -->
                    <div class="flex gap-4 text-lg opacity-80 pl-4">
                        <i class="${data.pinned ? 'ph-fill' : 'ph'} ph-push-pin transition-colors active:scale-90" 
                           style="color: ${data.pinned ? 'var(--color-warning)' : 'var(--text-muted)'};" 
                           onclick="togglePin(${data.id}, event)"></i>
                        <i class="ph ph-trash transition-colors active:scale-90 hover:text-red-400" 
                           style="color: var(--text-muted);" 
                           onclick="deleteItem(${data.id}, event)"></i>
                    </div>
                </div>
                <h4 class="text-[14px] font-bold mt-1.5 leading-snug pr-2">${data.title}</h4>
                <p class="text-[11px] opacity-60 mt-1 flex items-center gap-1.5 font-bold"><i class="ph ph-link text-sm"></i> ${data.source}</p>
            `;
            list.appendChild(itemEl);
        });
    }

                    <span class="text-[10px] uppercase font-bold tracking-widest" style="color: var(--text-muted);">🎯 落地场景与风险阻碍</span>
                    <span class="text-[13px] leading-relaxed opacity-90">${data.useCase}</span>
                    <span class="text-[13px] leading-relaxed mt-1" style="color: var(--color-warning);">${data.friction}</span>
                </div>

                <div class="p-3.5 rounded-xl border mt-2" style="border-color: rgba(129, 161, 193, 0.4); background-color: rgba(129, 161, 193, 0.08);">
                    <span class="font-bold block mb-2 text-xs flex items-center gap-1.5 tracking-wider" style="color: var(--color-primary);">
                        <i class="ph-fill ph-target"></i> 战略级行动建议
                    </span>
                    <p class="text-[13px] leading-relaxed opacity-90">${data.action}</p>
                </div>

                    style="background-color: var(--color-primary); color: var(--bg-base);"

            </button>
        `;

        // 初始化卡片样式
        cardEl.style.transform = 'translate(0px, 0px) rotate(0deg)';
        cardEl.style.opacity = '1';

        // 显示遮罩层
        document.getElementById('modal-library-detail').classList.remove('opacity-0', 'pointer-events-none');

        // 使用克隆节点重置事件监听器，避免多次绑定
        const newCardEl = cardEl.cloneNode(true);
        cardEl.parentNode.replaceChild(newCardEl, cardEl);

        makeLibraryCardDraggable(newCardEl);
    }

    function closeLibraryDetail() {
        const modal = document.getElementById('modal-library-detail');
        modal.classList.add('opacity-0', 'pointer-events-none');
    }

    // 专为智库卡片定制的左右滑动关闭逻辑
    function makeLibraryCardDraggable(card) {
        let isDragging = false, startX = 0, startY = 0, currentX = 0, currentY = 0;
        const badgeLike = card.querySelector('.badge.like');
        const badgeNope = card.querySelector('.badge.nope');

        const startDrag = (e) => {
            if (e.target.closest('button')) return;
            isDragging = true;
            card.classList.add('dragging');
            startX = e.type.includes('mouse') ? e.pageX : e.touches[0].pageX;
            startY = e.type.includes('mouse') ? e.pageY : e.touches[0].pageY;
        };

        const onDrag = (e) => {
            if (!isDragging) return;
            const x = e.type.includes('mouse') ? e.pageX : e.touches[0].pageX;
            const y = e.type.includes('mouse') ? e.pageY : e.touches[0].pageY;
            currentX = x - startX; currentY = y - startY;
            
            card.style.transform = `translate(${currentX}px, ${currentY}px) rotate(${currentX * 0.04}deg)`;

            // 无论是向左还是向右，都显示灰色的 CLOSE 徽章
            if (currentX > 0) {
                badgeLike.style.opacity = Math.min(currentX / 100, 1);
                badgeNope.style.opacity = 0;
            } else {
                badgeNope.style.opacity = Math.min(Math.abs(currentX) / 100, 1);
                badgeLike.style.opacity = 0;
            }
        };

        const stopDrag = () => {
            if (!isDragging) return;
            isDragging = false;
            card.classList.remove('dragging');

            if (Math.abs(currentX) > 100) {
                // 触发滑出并关闭
                const sign = currentX > 0 ? 1 : -1;
                card.style.transform = `translate(${sign * window.innerWidth}px, ${currentY}px) rotate(${sign * 30}deg)`;
                card.style.opacity = 0;
                setTimeout(() => closeLibraryDetail(), 300);
            } else {
                // 滑动距离不够，弹回原位
                card.style.transform = `translate(0px, 0px) rotate(0deg)`;
                badgeLike.style.opacity = 0; badgeNope.style.opacity = 0;
            }
        };

        card.addEventListener('mousedown', startDrag);
        card.addEventListener('touchstart', startDrag, {passive: true});
        document.addEventListener('mousemove', onDrag);
        document.addEventListener('touchmove', onDrag, {passive: true});
        document.addEventListener('mouseup', stopDrag);
        document.addEventListener('touchend', stopDrag);
    }

</script>
